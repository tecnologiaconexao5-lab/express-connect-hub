       valor_frete, valor_repasse, (metadata->>'numeroOC') AS numero_oc,
       (metadata->>'origem') AS origem, created_at
  FROM public.alocacoes_escala WHERE id = :alocacao_id;

-- 2c/2d) Item e faturamento derivado
SELECT fi.id, fi.fatura_id, fi.os_id, fi.alocacao_id, fi.romaneio, fi.status,
       fi.valor_total, f.numero, f.status AS fatura_status
  FROM public.fatura_itens fi JOIN public.faturas f ON f.id=fi.fatura_id
 WHERE fi.alocacao_id = :alocacao_id;
-- (derivacao da Conferencia: f.status='emitida' => 'faturado';
--  rascunho/aguardando/aprovada => 'rascunho' => sai da fila default "nao_faturado")

-- 3/5/7/8) Fatura
SELECT id, status, numero, numero_cliente, valor_bruto, valor_total, vencimento,
       data_emissao, conta_receber_id, tenant_id, cliente_id, criado_em, atualizado_em
  FROM public.faturas WHERE id = :fatura_id;

-- 3/5b/7b/8/9) Anti-duplicidade de AR
SELECT count(*)::int AS n FROM public.financeiro_receber WHERE fatura_id = :fatura_id;  -- 0 antes; 1 apos emitir

-- 8/8b/10) AR
SELECT id, status, valor, valor_liquido, cliente_id, data_vencimento, data_emissao,
       fatura_id, numero, is_teste, created_at
  FROM public.financeiro_receber WHERE fatura_id = :fatura_id;

-- 10) Ordenacao (Mais recente) no mes do VENCIMENTO do AR
SELECT id, created_at, data_vencimento,
       row_number() OVER (ORDER BY created_at DESC NULLS LAST) AS pos
  FROM public.financeiro_receber
 WHERE to_char(data_vencimento,'YYYY-MM') = to_char(:vencimento::date,'YYYY-MM')
 ORDER BY created_at DESC NULLS LAST LIMIT 20;

-- 12) Mapa de dependencias (NAO deletar)

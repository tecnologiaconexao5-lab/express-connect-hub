PARTE A — CERTIFICADO / mTLS
================================================================================
A.1 RUNTIME REAL DO PROJETO
  - supabase/config.toml -> [edge_runtime] enabled = true, deno_version = 2.
  - Edge Functions do projeto sao Deno (Supabase Edge Runtime), hospedadas.
  - Nao ha versao de edge-runtime fixada no repo; a versao e a do hosted.

A.2 COMPROVACAO TECNICA (API/runtime)
  - `fetch()` do Deno NAO aceita .p12/.pfx/PKCS#12 nem DER. Somente PEM.
    (Deno Network API: TlsCertifiedKeyPem exige PEM; "DER is not compatible".)
  - `Deno.createHttpClient({ cert, key, caCerts })` e o primitivo para mTLS de
    SAIDA. Foi EXPOSTO no edge-runtime pelo PR #515 "expose Deno.createHttpClient"
    (merge 18/03/2025, marco Deno 2.1 / issue #208 "query a TLS certificate-
    protected server"). Logo, em Deno 2 o primitivo existe.
  - Nomes corretos das opcoes: `cert` e `key`.
    Nomes do ecossistema Node (`certChain`/`privateKey`) sao IGNORADOS
    silenciosamente (cliente conecta SEM certificado) — NAO usar.
  - `keyFormat`/chave: Deno 2 (rustls 0.23+) exige chave PKCS#8
    (`-----BEGIN PRIVATE KEY-----`). Chave PKCS#1 (`BEGIN RSA PRIVATE KEY`)
    pode DERRUBAR o isolate (regressao reportada no edge-runtime 1.76.0 /
    Deno 2.1.4, set/2026 — ver evidencias abaixo).
  - `Deno.connectTls({ cert, key })` tambem suporta mTLS, mas foi citado como
    travando (>20s) no relato de regressao; preferir `createHttpClient`.

A.3 NECESSARIO (decisao)
  - [SIM] Converter .p12 -> PEM cert + PEM key (PKCS#8). (.p12 base64 nao basta.)
  - [NAO] Senha do .p12 em runtime: a senha so e usada na conversao, FORA do
    runtime. O secret entregue ao Deno ja e PEM sem senha.
  - [SIM] `Deno.createHttpClient` (cert/key) — mecanismo suportado.
  - [NAO] `.p12` direto no Secret como fonte de runtime.
  - [NAO] fetch simples: fetch padrao nao apresenta certificado de cliente.

A.4 FORMA MAIS SEGURA DE ARMAZENAR (sem valores)
  - Client ID / Client Secret / cert / key: SOMENTE em Supabase Secrets (backend).
  - Certificado em PEM (X.509) e chave em PEM PKCS#8, codificados em base64.
  - Converter o .p12 FORA do repositorio, em maquina controlada.
  - NAO copiar .p12 para o projeto. .gitignore ja cobre *.p12/*.pfx/*.pem/*.key.
  - Limites de secret (docs Supabase): 100 secrets/projeto, 48 KiB por secret —
    cert+key (poucos KB) cabem com folga.

-- ============================================================================
-- efi_cobrancas (A) — vinculo OBRIGATORIO fatura_id + receber_id + ambiente
-- ============================================================================
CREATE TABLE IF NOT EXISTS public.efi_cobrancas (
    id                uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id         uuid NOT NULL DEFAULT current_tenant_id_safe(),
    ambiente          text NOT NULL DEFAULT 'homologacao',
    fatura_id         uuid NOT NULL REFERENCES public.faturas(id) ON DELETE RESTRICT,
    receber_id        uuid NOT NULL REFERENCES public.financeiro_receber(id) ON DELETE RESTRICT,
    charge_id         text,
    txid              text,
    nosso_numero      text,
    status            text NOT NULL DEFAULT 'criada',
    valor             numeric(18,2) NOT NULL,
    vencimento        date,
    linha_digitavel   text,
    codigo_barras     text,
    pdf_url           text,
    barcode_url       text,
    pix_copia_cola    text,
    pix_qrcode_url    text,
    bolix             jsonb,
    metadata          jsonb NOT NULL DEFAULT '{}'::jsonb,   -- sanitizado
    criado_em         timestamptz NOT NULL DEFAULT now(),
    atualizado_em     timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT ck_efi_cobrancas_ambiente CHECK (ambiente IN ('homologacao','producao')),
    CONSTRAINT ck_efi_cobrancas_status CHECK (status IN (
      'criada','registrada','paga','cancelada','expirada','falhou','baixada'))
);
-- Anti-duplicidade (1 cobranca ATIVA por recebivel / por fatura):
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_cobrancas_receber_ativa
  ON public.efi_cobrancas (receber_id)
  WHERE status NOT IN ('cancelada','expirada','baixada');
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_cobrancas_charge
  ON public.efi_cobrancas (ambiente, charge_id) WHERE charge_id IS NOT NULL;
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_cobrancas_txid
  ON public.efi_cobrancas (ambiente, txid) WHERE txid IS NOT NULL;
CREATE INDEX IF NOT EXISTS idx_efi_cobrancas_fatura ON public.efi_cobrancas (fatura_id);
CREATE INDEX IF NOT EXISTS idx_efi_cobrancas_receber ON public.efi_cobrancas (receber_id);
CREATE INDEX IF NOT EXISTS idx_efi_cobrancas_status ON public.efi_cobrancas (status);

-- ============================================================================
-- efi_pix_pagamentos (B) — vinculo OBRIGATORIO pagamento_prestador_id + ambiente
-- ============================================================================
CREATE TABLE IF NOT EXISTS public.efi_pix_pagamentos (
    id                     uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id              uuid NOT NULL DEFAULT current_tenant_id_safe(),
    ambiente               text NOT NULL DEFAULT 'homologacao',
    pagamento_prestador_id uuid NOT NULL REFERENCES public.pagamentos_prestadores(id) ON DELETE RESTRICT,
    idempotency_key        text NOT NULL,
    end_to_end_id          text,
    status                 text NOT NULL DEFAULT 'preparado',
    valor                  numeric(18,2) NOT NULL,
    chave_pix_destino      text,
    referencia             text,
    comprovante_url        text,
    baixa_id               uuid REFERENCES public.financeiro_baixas(id) ON DELETE SET NULL,
    erro                   text,
    metadata               jsonb NOT NULL DEFAULT '{}'::jsonb,
    criado_em              timestamptz NOT NULL DEFAULT now(),
    atualizado_em          timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT ck_efi_pix_ambiente CHECK (ambiente IN ('homologacao','producao')),
    CONSTRAINT ck_efi_pix_status CHECK (status IN (
      'preparado','enviado','liquidado','falhou','cancelado','devolvido'))
);
-- Anti-duplicidade:
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_pix_idempotency
  ON public.efi_pix_pagamentos (idempotency_key);
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_pix_e2e
  ON public.efi_pix_pagamentos (ambiente, end_to_end_id) WHERE end_to_end_id IS NOT NULL;
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_pix_pagamento_ativo
  ON public.efi_pix_pagamentos (pagamento_prestador_id)
  WHERE status NOT IN ('falhou','cancelado','devolvido');
CREATE INDEX IF NOT EXISTS idx_efi_pix_pagamento ON public.efi_pix_pagamentos (pagamento_prestador_id);
CREATE INDEX IF NOT EXISTS idx_efi_pix_status ON public.efi_pix_pagamentos (status);

-- ============================================================================
-- efi_eventos — webhooks (idempotencia de ingestao)
-- ============================================================================
CREATE TABLE IF NOT EXISTS public.efi_eventos (
    id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       uuid,
    ambiente        text NOT NULL DEFAULT 'homologacao',
    tipo            text NOT NULL,
    efi_id          text,
    idempotency_key text NOT NULL,
    payload         jsonb NOT NULL DEFAULT '{}'::jsonb,   -- sanitizado
    processado      boolean NOT NULL DEFAULT false,
    erro            text,
    recebido_em     timestamptz NOT NULL DEFAULT now(),
    processado_em   timestamptz,
    CONSTRAINT ck_efi_eventos_ambiente CHECK (ambiente IN ('homologacao','producao'))
);
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_eventos_idempotency
  ON public.efi_eventos (idempotency_key);
CREATE UNIQUE INDEX IF NOT EXISTS uq_efi_eventos_externo
  ON public.efi_eventos (ambiente, tipo, efi_id) WHERE efi_id IS NOT NULL;
CREATE INDEX IF NOT EXISTS idx_efi_eventos_processado ON public.efi_eventos (processado);

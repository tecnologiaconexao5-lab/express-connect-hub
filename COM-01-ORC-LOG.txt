FORMULA DO ORCAMENTO (orcamentoCalculo.ts):
  valorFinal = valorBase + adicionais + pedagio + kmExcedente + ajudante
               + adValorem + gris + devolucao + reentrega - descontos
  adValorem  = valorDeclarado * 0,30%   (0 p/ Amazon Logistica / Industrias ABC)
  gris       = valorDeclarado * 0,15%   (idem)
  lucroEstimado  = valorFinal - custoEstimado
  margemEstimada = lucroEstimado / valorFinal * 100

REGRA DO CLIENTE (buscarTabelaValoresCliente + engine):
  valorCliente = valor_base + max(0, distancia_km - franquia_km) * valor_km_excedente
  Hierarquia de selecao (7 niveis):
    1 cliente_id | 2 contrato | 3 cliente+veiculo+vigencia | 4 cliente+veiculo
    5 universal+veiculo+vigencia | 6 universal+veiculo | 7 fallback
  Parametros usados: distancia (Mapbox), peso/cubagem (recomendacao de veiculo),
  tipo de veiculo, valor declarado (adValorem/gris), prazo (faixas).

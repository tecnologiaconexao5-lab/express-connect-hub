    CREATE OR REPLACE FUNCTION fn_audit_liberado_app()
    RETURNS trigger AS $$
    BEGIN
      IF NEW.liberado_app IS DISTINCT FROM OLD.liberado_app THEN
        INSERT INTO liberado_app_audit (prestador_id, liberado, motivado_por)
        VALUES (NEW.id, NEW.liberado_app, COALESCE(NEW.updated_by, 'sistema'));

      RETURN NEW;
    END;

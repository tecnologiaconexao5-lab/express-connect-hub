-- 7/8/12) Alocação (Conferência)
SELECT id, tenant_id, status, finalizada, data_finalizacao,
       cliente_id, prestador_id, valor_frete, valor_repasse,
       (metadata->>'numeroOC') AS numero_oc, (metadata->>'origem') AS origem
  FROM public.alocacoes_escala WHERE id = :alocacao_id;

-- 12) Se o item já está em fatura ativa (inelegível a novo faturamento)
SELECT fi.id, fi.status AS item_status, f.status AS fatura_status, f.numero
  FROM public.fatura_itens fi JOIN public.faturas f ON f.id = fi.fatura_id
 WHERE fi.alocacao_id = :alocacao_id AND fi.status = 'ativo';

-- 15/17/19/22) Fatura (status/valores/AR)
SELECT id, status, numero, numero_cliente, conta_receber_id,
       valor_total, vencimento, data_emissao, tenant_id
  FROM public.faturas WHERE id = :fatura_id;

-- 20/22) Conta a Receber da fatura (deve ser 0 na validacao/aprovacao; 1 na emissao)
SELECT count(*)::int AS n FROM public.financeiro_receber WHERE fatura_id = :fatura_id;
SELECT id, status, valor, valor_liquido, cliente_id, data_vencimento, fatura_id, numero
  FROM public.financeiro_receber WHERE fatura_id = :fatura_id;

-- 12/22) Garantia de não-duplicidade (máx. 1 AR por fatura e 1 fatura ativa por OS)
SELECT fatura_id, count(*)::int AS n FROM public.financeiro_receber
 WHERE fatura_id IS NOT NULL GROUP BY fatura_id HAVING count(*) > 1;

-- Obrigações de PRESTADOR criadas automaticamente pelo caso (mapear, NÃO alterar)
SELECT 'financeiro_pagar' AS tabela, id::text, status, valor FROM public.financeiro_pagar WHERE os_id = :os_id
UNION ALL
SELECT 'pagamentos_prestadores', id::text, status_pagamento, valor_liquido FROM public.pagamentos_prestadores WHERE os_id = :os_id
UNION ALL
SELECT 'composicao_financeira_os', os_id::text, NULL, valor_prestador FROM public.composicao_financeira_os WHERE os_id = :os_id;

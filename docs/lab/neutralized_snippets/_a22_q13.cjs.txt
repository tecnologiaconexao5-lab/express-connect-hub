const { Client } = require('pg');
const DB_URL = process.env.DBURL;
(async () => {
  const c = new Client({ connectionString: DB_URL });
  await c.connect();
  const n = await c.query("SELECT id, prestador_id, titulo, mensagem, tipo, referencia_id, event_type, lida, created_at FROM public.portal_notificacoes WHERE tipo = 'operacao_mensagem' AND referencia_id = 'ca5fe225-7114-42da-9997-c3d466e5df52' ORDER BY created_at DESC LIMIT 5");
  console.log('NOTIFICACOES:', JSON.stringify(n.rows, null, 2));
  await c.end();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });

﻿const { Client } = require('pg');
const DB_URL = process.env.DBURL;
(async () => {
  const c = new Client({ connectionString: DB_URL });
  await c.connect();
  const r = await c.query("SELECT p.proname AS nome, pg_get_function_arguments(p.oid) AS args, pg_get_function_result(p.oid) AS ret FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace WHERE n.nspname = 'public' AND p.proname ILIKE '%mensagens_nao_lidas%'");
  console.log(JSON.stringify(r.rows, null, 2));
  await c.end();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });

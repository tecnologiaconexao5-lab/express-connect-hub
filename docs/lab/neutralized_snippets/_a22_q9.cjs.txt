const { Client } = require('pg');
const DB_URL = process.env.DBURL;
(async () => {
  const c = new Client({ connectionString: DB_URL });
  await c.connect();
  const cols = await c.query("SELECT column_name FROM information_schema.columns WHERE table_schema='public' AND table_name='tenant_usuarios' ORDER BY ordinal_position");
  console.log('COLUNAS tenant_usuarios:', cols.rows.map(x => x.column_name).join(', '));
  const tr = await c.query("SELECT * FROM public.tenant_usuarios LIMIT 10");
  console.log('TENANT_USUARIOS:', JSON.stringify(tr.rows, null, 2));
  await c.end();
})().catch(e => { console.error('ERR', e.message); process.exit(1); });

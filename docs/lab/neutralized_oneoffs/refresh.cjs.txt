  const novaTabela = {

    prestador_documento: '00000000000',
    periodo_inicio: '2025-01-01',
    periodo_fim: '2025-01-31',
    quantidade_os: 1,
    valor_servicos: 100,
    valor_reembolsos: 0,

    created_at: new Date().toISOString(),
